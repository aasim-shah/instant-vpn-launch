Getting Started

Time to first API call: ~5 minutes

1. Add the Dependency

build.gradle.kts — Project
repositories {
    google()
    mavenCentral()
    maven { url = uri("https://maven.fyreway.com/releases") }
}
build.gradle.kts — App module
dependencies {
    // FyreWay VPN SDK
    implementation("com.fyreway.vpn:fyreway-vpn-sdk:1.1.0")

    // Required
    implementation("org.jetbrains.kotlinx:kotlinx-coroutines-android:1.7.3")
    implementation("androidx.lifecycle:lifecycle-viewmodel-ktx:2.6.2")
    implementation("androidx.lifecycle:lifecycle-runtime-ktx:2.6.2")
}

If you're using an AAR file directly:

dependencies {
    implementation(files("libs/fyreway-vpn-sdk-release.aar"))
}

2. Update AndroidManifest.xml

AndroidManifest.xml
<manifest xmlns:android="http://schemas.android.com/apk/res/android">

    <!-- Required -->
    <uses-permission android:name="android.permission.INTERNET" />
    <uses-permission android:name="android.permission.ACCESS_NETWORK_STATE" />

    <application
        android:name=".MyApplication"
        ... >
        <!-- Your activities -->
    </application>
</manifest>

3. Initialize the SDK

Initialize once in your Application.onCreate(). Never initialize in an Activity or Fragment.

Kotlin
class MyApplication : Application() {
    override fun onCreate() {
        super.onCreate()

        FyreWaySDK.initialize {
            baseUrl = "https://your-api-server.com"    // Required
            enableLogging = BuildConfig.DEBUG            // Disable in production
            logLevel = LogLevel.BASIC                    // NONE, BASIC, HEADERS, BODY
            connectTimeout = 30_000L
            readTimeout = 30_000L
            writeTimeout = 30_000L
            maxRetries = 3
            cachePolicy = CachePolicy.CACHE_FIRST
            analyticsConfig = AnalyticsConfig(
                enabled = true,
                batchSize = 50,
                flushInterval = 60_000L
            )
        }
    }
}
Java
public class MyApplication extends Application {
    @Override
    public void onCreate() {
        super.onCreate();

        FyreWayConfig config = new FyreWayConfig.Builder()
            .baseUrl("https://your-api-server.com")
            .enableLogging(BuildConfig.DEBUG)
            .logLevel(LogLevel.BASIC)
            .build();

        FyreWaySDK.initialize(config);
    }
}
Important: baseUrl is the only required parameter. All others have sensible defaults. Never hardcode your production URL — use BuildConfig fields.

4. Activate the Device

Device activation registers this device with the backend and returns authentication credentials. Call it once on first launch — the SDK stores credentials securely and reuses them.

sequenceDiagram
    participant App as Your App
    participant SDK as FyreWay SDK
    participant Store as Secure Storage
    participant API as Backend API
    App->>SDK: activateDevice("1.0.0")
    SDK->>API: POST /devices/activate
    API-->>SDK: deviceId, token, refreshToken
    SDK->>Store: Store credentials
    SDK->>SDK: Start proactive token refresh
    SDK-->>App: VpnResult.Success(DeviceCredentials)
Kotlin
class MainViewModel : ViewModel() {
    private val sdk = FyreWaySDK.getInstance()

    fun activate() {
        viewModelScope.launch {
            when (val result = sdk.activateDevice(appVersion = "1.0.0")) {
                is VpnResult.Success -> {
                    val deviceId = result.data.deviceId
                    // Device is ready — you can now call any SDK API
                }
                is VpnResult.Error -> {
                    val message = ErrorMessageMapper.getMessage(result.exception)
                    // Show user-friendly error
                }
            }
        }
    }
}
Java
FyreWayJavaClient client = FyreWayJavaClient.initialize();

client.activateDevice("1.0.0", new FyreWayCallback<DeviceCredentials>() {
    @Override
    public void onSuccess(DeviceCredentials credentials) {
        // Device is ready
    }

    @Override
    public void onError(Exception exception, Integer statusCode) {
        // Handle error
    }
});

5. Check Authentication State

val sdk = FyreWaySDK.getInstance()

if (sdk.isAuthenticated()) {
    // Proceed with API calls
} else {
    // Need to activate device first
    sdk.activateDevice("1.0.0")
}

6. Token Lifecycle

The SDK manages token refresh automatically. After activation, a background job checks every 5 minutes and refreshes the token before it expires.

graph LR
    A[activateDevice] --> B[Token stored]
    B --> C{Expiring within 24h?}
    C -- Yes --> D[refreshToken]
    D --> B
    C -- No --> E[Wait 5 min]
    E --> C
MethodWhen to use
startProactiveTokenRefresh()Called automatically after activation. Call manually only if you previously stopped it.
stopProactiveTokenRefresh()Call before logout or when you no longer need the SDK.
refreshToken()Manual one-time refresh. Rarely needed — the SDK handles this.

7. Logout

sdk.logout()  // Clears all credentials and stops token refresh

After logout, you must call activateDevice() again before using any API.

Minimal Working Example

Kotlin — Complete
// Application.kt
class MyApp : Application() {
    override fun onCreate() {
        super.onCreate()
        FyreWaySDK.initialize {
            baseUrl = "https://your-api-server.com"
        }
    }
}

// MainViewModel.kt
class MainViewModel : ViewModel() {
    private val sdk = FyreWaySDK.getInstance()

    init { activate() }

    private fun activate() = viewModelScope.launch {
        if (!sdk.isAuthenticated()) {
            sdk.activateDevice("1.0.0")
        }
    }

    fun getServers() = viewModelScope.launch {
        when (val result = sdk.getServers()) {
            is VpnResult.Success -> {
                val servers = result.data.servers
                // Display servers to user
            }
            is VpnResult.Error -> { /* Handle error */ }
        }
    }
}
